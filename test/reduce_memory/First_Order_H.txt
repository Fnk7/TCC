dist3(p) = dist(p) * dist(p) * dist(p)
pf1(p) = ptab(p) * (sz * dir(p) * dist3(p))
first_order_sigama(p) = sum((c1), first_order_gradient_rho(p,c1) * gradient_rho(p,c1))
pf2(p,c1) = ptab(p) * (gradient_rho(p,c1) * (v2rhosigma(p) * first_order_rho(p) + v2sigma2(p) * first_order_sigama(p)) + first_order_gradient_rho(p,c1) * vsigma(p))
pf3(p) = ptab(p) * (v_hartree_gradient(p) + v2rho2(p) * first_order_rho(p) + v2rhosigma(p) * first_order_sigama(p))
pf4_1(p) = ptab(p)
pf4_2(p) = ptab(p) * vrho(p)
pr4_3(p,c1) = partition_tab(p) * vsigma(p) * gradient_rho(p,c1)
pr5_1(p,c1) = partition_tab(p) * vsigma(p) * gradient_rho(p,c1)
pf5_2(p) = ptab(p)
first_order_wave(p,i) = zeroif((j_atom != atom_of(i)), gradient_basis_wave(p,c2,i))
first_order_kinetic_wave(p,i) = zeroif((j_atom != atom_of(i)), kinetic_gradient_basis_wave(p,c2,i))
first_order_gradient_wave(p,i,c1) = zeroif((j_atom != atom_of(i)), hessian_basis_wave(p,c1,c2,i))
ct1(p,i) = pf1(p) * wave(p,i)
ct2(p,i) = sum((c1), pf2(p,c1) * gradient_basis_wave(p,c1,i))
ct3(p,i) = pf3(p) * wave(p,i)
ct4(p,i) = pf4_1(p) * H_times_psi(p,i) + pf4_2(p) * wave(p,i)  + sum((c1), pr4_3(p,c1) * gradient_basis_wave(p,c1,i))
ct5(p,i) = sum((c1), pr5_1(p,c1) * first_order_gradient_wave(p,i,c1)) + pf5_2(p) * first_order_kinetic_wave(p,i)
cs1(p,i) = ct1(p,i) + ct2(p,i) + ct3(p,i) + ct5(p,i)
cs2(p,i) = ct4(p,i)
H(i,j) = sum((p), wave(p,i) * cs1(p,j) + cs1(p,i) * wave(p,j))
H(i,j) = H(i,j) + sum((p), first_order_wave(p,i) * cs2(p,j) + cs2(p,i) * first_order_wave(p,j))
keep H
